class Customer {
    String name;
    String address;
}

class Restaurant {
    String name;
    String location;
}
class DeliveryBoy {
    String name;
    String phone;
}
class Order {
    Customer customer;
    Restaurant restaurant;
    DeliveryBoy deliveryBoy;
    Order(Customer c, Restaurant r, DeliveryBoy d) {
        customer = c;
        restaurant = r;
        deliveryBoy = d;
    }
    void showOrderDetails() {
        print("Customer: " + customer.name);
        print("Restaurant: " + restaurant.name);
        print("Delivery By: " + deliveryBoy.name);
    }
}
Order order = new Order(new Customer("Ravi", "Delhi"),
                        new Restaurant("SpiceHub", "Connaught Place"),
                        new DeliveryBoy("Amit", "9876543210"));
order.showOrderDetails();

class Book {
    String title;
    int year;
    double rating;
}

List<Book> books = List.of(
    new Book("Book A", 2016, 4.6),
    new Book("Book B", 2014, 4.8),
    new Book("Book C", 2018, 4.9)
List<Book> filtered = books.stream()
    .filter(b -> b.year > 2015 && b.rating > 4.5)
    .collect(Collectors.toList());
filtered.forEach(b -> print(b.title));
