Stack<String> undoStack = new Stack();
String currentText = "";

function type(text) {
    undoStack.push(currentText);
    currentText += text;
}

function undo() {
    if (!undoStack.isEmpty()) {
        currentText = undoStack.pop();
    }
}

function showText() {
    print(currentText);
}

    .stream()
    .max((a, b) -> a.getValue() - b.getValue())
