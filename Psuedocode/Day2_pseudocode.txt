Question -> 1. enum TrafficLight {
    RED, GREEN, YELLOW
}

class SignalController {
    TrafficLight current = RED;

    void switchLight() {
        if (current == RED) current = GREEN;
        else if (current == GREEN) current = YELLOW;
        else current = RED;
    }

    void showLight() {
        print("Current Light: " + current);
    }
}


SignalController signal = new SignalController();
signal.showLight();  
signal.switchLight(); 
signal.showLight();

class Doctor {

    String specialization;
}

departmentMap.put("Cardiology", List.of(new Doctor("Dr. A", "Heart")));
departmentMap.put("Neurology", List.of(new Doctor("Dr. B", "Brain")));
departmentMap.put("Orthopedics", List.of(new Doctor("Dr. C", "Bones")));
departmentMap.values()
    .stream()
    .flatMap(list -> list.stream())
    .filter(doc -> doc.specialization.equals("Heart"))
